add_library(nockvm_display STATIC monitor_info.cpp)
target_include_directories(nockvm_display PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_link_libraries(nockvm_display PUBLIC nockvm_common)

if(WIN32)
  target_link_libraries(nockvm_display PUBLIC user32 gdi32)
else()
  find_package(X11 REQUIRED)
  if(NOT X11_Xrandr_FOUND)
    message(FATAL_ERROR "XRandr development files not found (libxrandr-dev / libXrandr-devel)")
  endif()
  target_include_directories(nockvm_display PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(nockvm_display PUBLIC ${X11_LIBRARIES} ${X11_Xrandr_LIB})
endif()
